        * Solve possible compilation problems
            Problem:
            - object files cannot be linked against each other due to mangeling of the c++ compiler
            - functions may be overloaded in c++ (same name, different signature = arguments)
            - also symbols may have the same name in different namespaces
            - mangeling does encode symbols in a way that they cannot be mixed up
            - this may also happen to symbols in cyboi c or header files so that they cannot be found within the c shared library
            Solution:
            - insert special compiler directives into the cyboi c header files
            - in order to DISABLE mangeling of the compiler at these places

Sleep
_____

        * usleep() works with microseconds.
        * In windows for getting microsecond precesion you should use QueryPerformanceCounter() winapi function.
        * Here you can find how get that precesion using it.

            #include <windows.h>
            void uSleep(int waitTime) {
                __int64 time1 = 0, time2 = 0, freq = 0;
                QueryPerformanceCounter((LARGE_INTEGER *) &time1);
                QueryPerformanceFrequency((LARGE_INTEGER *)&freq);
                do {
                    QueryPerformanceCounter((LARGE_INTEGER *) &time2);
                } while((time2-time1) < waitTime);
            }

            (Platform independence)
            sleep() is not platform-independent.
            usleep() is not platform-independent either. Also, it takes it's
            argument in microseconds, not milliseconds. usleep(1000) sleeps for
            1000 microseconds, that's 1 millisecond, not 1 second. You can't use
            usleep() to sleep for 1 second -- usleep(1000000) is not valid. See
            http://linux.die.net/man/3/usleep .
            Neither sleep() nor usleep() are defined by C++.
            There is no platform-independent way to sleep in milliseconds (or in any unit).
            You might find something in boost (e.g. boost::thread::sleep()) or some other library.
            If you really have protability issue, try using "select". It will work on POSIX systems.
            Es geht auch so (auf verschiedenen Unix-Derivaten), C++ Example:

            #define TRUE          1
            #define NONE          0                    (oder auch NULL)
            #define ERRINTR(e)   ((e) == EINTR)

            int inputWait(msec)
            int msec;       /*  I:  time to wait [msec] */
                                    /* ret: <0 if any error */
            /* procedure waits msec */
            {
                struct timeval tv;  /* help variable */
                int errno2;     /* help variable */
                int ret;        /* return variable */
                ret = 0;

                if (msec >= 1000) {
                    tv.tv_sec = msec / 1000;
                    tv.tv_usec = (msec % 1000)  * 1000;
                } else {
                    tv.tv_sec = 0;
                    tv.tv_usec = msec * 1000;
                }

                while (TRUE) {
                    /* do looping while interrupt by signal catching */
                    if (select(0, NONE, NONE, NONE, &tv) < 0) {
                        errno2 = errno;
                        XBUG(DBG_SELECT) {

                            fprintf(dbgfile,">>>inputWait(): INTR at select(), errno=%d\n",errno); fflush(dbgfile);
                        }
                        if (! ERRINTR(errno2)) {
                            break;
                        }
                        /* else: EINTR => return looping */
                    } else {
                        break;
                    }
                }
                return(ret);
            }

            Der Aufruf select() macht ein blockierendes Warten, bis die Zeitscheibe abgelaufen ist, also analog sleep().
            Kann man natürlich auch für usec anpassen ;-)

/**
 * Casts the source of type double to the destination of type integer.
 *
 * CAUTION! The fractional/decimal post-point value gets lost.
 *
 * CAUTION! Floating point numbers are precision-dependent and have an inaccuracy.
 * For example, the decimal number 9.2 is never exactly equal to 9.2:
 * - 32-bit "single precision" float: 9.19999980926513671875
 * - 64-bit "double precision" float: 9.199999999999999289457264239899814128875732421875
 * https://stackoverflow.com/questions/21895756/why-are-floating-point-numbers-inaccurate
 *
 * This has the effect that when casting for example the double value 4.0
 * which is internally represented as 3.999... to integer it becomes 3,
 * even though 4 would be expected.
 *
 * Therefore, the following preparation has to be applied BEFORE casting:
 * x > 0: add 0.5
 * x < 0: subtract 0.5
 * where x is the value to be casted.
 * https://stackoverflow.com/questions/9695329/c-how-to-round-a-double-to-an-int
 *
 * @param p0 the destination
 * @param p1 the source
 */
void cast_integer_double(void* p0, void* p1) {

    if (p1 != *NULL_POINTER_STATE_CYBOI_MODEL) {

        if (p0 != *NULL_POINTER_STATE_CYBOI_MODEL) {

            int* d = (int*) p0;

            log_message_terminated((void*) DEBUG_LEVEL_LOG_CYBOI_MODEL, (void*) L"Cast integer double.");
            //?? fwprintf(stdout, L"Debug: Cast integer double. *p1: %f\n", *((int*) p1));

            //
            // The temporary value to be prepared for casting.
            //
            // CAUTION! It IS necessary in order to AVOID manipulation
            // of the original source value.
            //
            double p = *NUMBER_0_0_DOUBLE_STATE_CYBOI_MODEL;
            // The comparison result.
            int r = *FALSE_BOOLEAN_STATE_CYBOI_MODEL;

            // Copy decimals (post point value) to that prepared for casting.
            copy_double((void*) &p, p1);

            compare_double_greater_or_equal((void*) &r, (void*) &p, (void*) NUMBER_0_0_DOUBLE_STATE_CYBOI_MODEL);

            if (r != *FALSE_BOOLEAN_STATE_CYBOI_MODEL) {

                calculate_double_add((void*) &p, (void*) NUMBER_0_5_DOUBLE_STATE_CYBOI_MODEL);

            } else {

                calculate_double_subtract((void*) &p, (void*) NUMBER_0_5_DOUBLE_STATE_CYBOI_MODEL);
            }

            // Cast double to integer value.
            *d = (int) p;

        } else {

            log_message_terminated((void*) ERROR_LEVEL_LOG_CYBOI_MODEL, (void*) L"Could not cast integer double. The destination is null.");
            fwprintf(stdout, L"Error: Could not cast integer double. The destination is null. p0: %i\n", p0);
        }

    } else {

        log_message_terminated((void*) ERROR_LEVEL_LOG_CYBOI_MODEL, (void*) L"Could not cast integer double. The source is null.");
        fwprintf(stdout, L"Error: Could not cast integer double. The destination is null. p1: %i\n", p1);
    }
}
